export type ThemeId = 'campus' | 'glass';
export type SaveState = 'clean' | 'dirty' | 'saving' | 'saved' | 'failed' | 'conflict' | 'readonly';
export type CommitResult = { ok: true; revision: number } | { ok: false; reason: 'readonly' | 'conflict' | 'validation' | 'capacity' | 'network'; message: string };
export type ArtworkSpec = { src: string; mobileSrc?: string; width: number; height: number; fit: 'contain' | 'cover'; focalPoint: [number, number]; decorative: boolean };
export interface ThemeDefinition { id: ThemeId; status: 'planned' | 'development' | 'ready'; artwork: Partial<Record<string, ArtworkSpec>>; capabilities: { blur: boolean; reducedMotion: boolean }; }
export interface EditorActions<T> { updateDraft(patch: Partial<T>): void; confirmReview(): void; commit(): Promise<CommitResult>; requestLeave(): Promise<boolean>; }
